#include <stdint.h>


#define VIDEO_MEMORY 0xB8000


void print_char(char c, uint8_t color) {
    uint16_t* video_memory = (uint16_t*)VIDEO_MEMORY;
    static int cursor = 0;
    
    video_memory[cursor] = (color << 8) | c;
    cursor++;
    

    if (cursor >= 80 * 25) {
        cursor = 0;
        for (int i = 0; i < 80 * 25; i++) {
            video_memory[i] = (color << 8) | ' ';
        }
    }
}


void print_string(const char* str, uint8_t color) {
    while (*str) {
        print_char(*str, color);
        str++;
    }
}


void kernel_main() {
    print_string("Hello from Gribarius OS!", 0x0F);  // Белый текст на черном фоне
    print_string("\nThis is a simple operating system", 0x0F);
    print_string("\nPress any key to continue...", 0x0F);

  
    while (1);
}


extern "C" void _start() {
    kernel_main();
}
